# Reverse proxy / load balancer devant les réplicas de l'API.
events { worker_connections 4096; }

http {
    access_log off;

    # DNS interne Docker : "api" résout vers toutes les réplicas (réévalué toutes les 5 s,
    # une instance arrêtée disparaît donc rapidement de la rotation).
    resolver 127.0.0.11 valid=5s ipv6=off;

    upstream api_pool {
        zone api_pool 64k;
        server api:8000 resolve max_fails=2 fail_timeout=5s;
        keepalive 32;
    }

    server {
        listen 80;

        location / {
            proxy_pass http://api_pool;
            proxy_http_version 1.1;
            proxy_set_header Connection "";
            proxy_set_header Host $host;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

            # Timeouts courts côté gateway et bascule vers une autre réplica si l'instance
            # est injoignable (connexion refusée / coupée). Jamais pour un POST (non idempotent).
            proxy_connect_timeout 500ms;
            proxy_read_timeout 5s;
            proxy_next_upstream error timeout http_502;
            proxy_next_upstream_tries 2;
            proxy_next_upstream_timeout 2s;
        }
    }
}
